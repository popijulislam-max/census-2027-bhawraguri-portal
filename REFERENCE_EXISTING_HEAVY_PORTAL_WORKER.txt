      if (path === "/api/health" && method === "GET") {

      // The D1 binding is required for all data operations.

      }

        return submitHLB(request, env.DB);
      }

        return adminLogin(request, env);
      }

      if (path === "/api/admin/logout" && method === "POST") {
        return adminLogout();
      }

      if (path === "/api/admin/session" && method === "GET") {
        return json({ success: true, authenticated: await isAdmin(request, env) });
      }

      // ---------------- ADMIN ONLY ----------------
      if (path.startsWith("/api/admin/")) {
        if (!(await isAdmin(request, env))) return unauthorized();

        if (path === "/api/admin/today") {
          return json(await dailyReport(env.DB, currentDate()));
        }

        if (path === "/api/admin/daily") {
          const date = clean(url.searchParams.get("date")) || currentDate();
          if (!validDate(date)) return json({ success:false, error:"Invalid report date." }, 400);
          return json(await dailyReport(env.DB, date));
        }

        if (path === "/api/admin/overall") {
          return json(await overallReport(env.DB));
        }

        if (path === "/api/admin/hlb") {
          const hlb = normalizeHLB(url.searchParams.get("hlb"));
          if (!hlb) return json({ success:false, error:"Please enter a valid HLB number from 0001 to 0178." }, 400);
          return json(await hlbDetails(env.DB, hlb));
        }

        if (path === "/api/admin/final-list") {
          return json({ success:true, total:FINAL_LIST.length, rows:FINAL_LIST });

        if (path === "/api/admin/reserve-list") {
          return json({ success:true, total:RESERVE_LIST.length, rows:RESERVE_LIST });

// ============================================================
// HELPERS
// ============================================================

function clean(v) {
  return String(v ?? "").trim();
}

function esc(s) {
  return String(s ?? "").replace(/[&<>"']/g, m => ({
    "&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#039;"
  }[m]));
}

function num(v) {
  const n = Number(v);
  return Number.isFinite(n) ? Math.max(0, Math.floor(n)) : 0;
}

function pct(a,b) {
  return b > 0 ? Math.round((a / b) * 10000) / 100 : 0;
}

function allHLBs() {
  return Array.from({length:TOTAL_HLB}, (_,i) => String(i+1).padStart(4,"0"));
}

function normalizeHLB(v) {
  const s = clean(v);
  if (!/^\d{1,4}$/.test(s)) return "";
  const n = Number(s);
  if (n < 1 || n > TOTAL_HLB) return "";
  return String(n).padStart(4,"0");
}

function validDate(s) {
  return /^\d{4}-\d{2}-\d{2}$/.test(s);
}

function currentDate() {
  const parts = new Intl.DateTimeFormat("en-CA", {
    timeZone: TIME_ZONE,
    year:"numeric",
    month:"2-digit",
    day:"2-digit"
  }).formatToParts(new Date());

  const p = {};
  for (const x of parts) p[x.type] = x.value;
  return p.year + "-" + p.month + "-" + p.day;
}

function displayDate(iso) {
  const d = new Date(iso + "T00:00:00+05:30");
  return new Intl.DateTimeFormat("en-GB", {
    timeZone:TIME_ZONE,
    day:"2-digit",
    month:"short",
    year:"numeric"
  }).format(d);
}

function json(data, status=200, extra={}) {
  return new Response(JSON.stringify(data), {
    status,
    headers:{
      "content-type":"application/json;charset=UTF-8",
      "cache-control":"no-store, no-cache, must-revalidate",
      ...extra
    }
  });
}

function html(body, status=200) {
  return new Response(body, {
    status,
    headers:{
      "content-type":"text/html;charset=UTF-8",
      "cache-control":"no-store, no-cache, must-revalidate"
    }
  });
}

function text(body,status=200) {
  return new Response(body, {
    status,
    headers:{"content-type":"text/plain;charset=UTF-8"}
  });
}

function unauthorized() {
  return json({success:false,error:"Admin login required."},401);
}

function csvCell(v) {
  return '"' + String(v ?? "").replace(/"/g,'""') + '"';
}

function csv(rows, filename) {
  const headers = rows.length ? Object.keys(rows[0]) : ["Message"];
  const lines = [headers.map(csvCell).join(",")];

  for (const row of rows) {
    lines.push(headers.map(h => csvCell(row[h])).join(","));
  }

  return new Response("\ufeff" + lines.join("\r\n"), {
    headers:{
      "content-type":"text/csv;charset=UTF-8",
      "content-disposition":'attachment; filename="' + filename + '"',
      "cache-control":"no-store"
    }
  });
}

async function initDB(db) {
  await db.prepare(`
    CREATE TABLE IF NOT EXISTS hlb_submissions(
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      report_date TEXT NOT NULL,
      enumerator_name TEXT NOT NULL,
      mobile TEXT NOT NULL,
      hlb TEXT NOT NULL,
      buildings INTEGER NOT NULL DEFAULT 0,
      houses INTEGER NOT NULL DEFAULT 0,
      synced INTEGER NOT NULL DEFAULT 0,
      status TEXT NOT NULL DEFAULT 'SUBMITTED',
      created_at TEXT NOT NULL
    )
  `).run();

  await db.prepare(`
    CREATE UNIQUE INDEX IF NOT EXISTS uq_hlb_report_date
    ON hlb_submissions(report_date, hlb)
  `).run();

  await db.prepare(`
    CREATE INDEX IF NOT EXISTS idx_hlb
    ON hlb_submissions(hlb)
  `).run();

  await db.prepare(`
    CREATE INDEX IF NOT EXISTS idx_report_date
    ON hlb_submissions(report_date)
  `).run();
}

async function publicToday(db) {
  const date = currentDate();

  const r = await db.prepare(`
    SELECT hlb FROM hlb_submissions
    WHERE report_date=?
    ORDER BY CAST(hlb AS INTEGER)
  `).bind(date).all();

  const submitted = new Set(
    (r.results || []).map(x => String(x.hlb).padStart(4,"0"))
  );

  return {
    success:true,
    report_date:date,
    report_date_display:displayDate(date),
    total_hlb:TOTAL_HLB,
    submitted_hlb:submitted.size,
    pending_hlb:TOTAL_HLB-submitted.size,
    hlbs:allHLBs().map(h => ({
      number:h,
      submitted:submitted.has(h)
    }))
  };
}

async function submitHLB(request, db) {
  let body;

  try {
    body = await request.json();
  } catch {
    return json({success:false,error:"Invalid JSON request."},400);
  }

  const hlb = normalizeHLB(body.hlb);
  const enumeratorName = clean(body.enumerator_name);
  const mobile = clean(body.mobile);
  const buildings = num(body.buildings);
  const houses = num(body.houses);
  const synced = num(body.synced);

  if (!hlb) return json({success:false,error:"Please select a valid HLB number from 0001 to 0178."},400);
  if (!enumeratorName) return json({success:false,error:"Enumerator Name is required."},400);
  if (!/^\d{10}$/.test(mobile)) return json({success:false,error:"Please enter a valid 10 digit mobile number."},400);
  if (synced > houses) return json({success:false,error:"Household Synced cannot be greater than Total Census Houses."},400);

  const official = sourceForHLB(hlb);

  // If the HLB exists in the official Final List, require the field entry
  // to match the official enumerator name and mobile number.
  if (official) {
    const sameName = clean(official.enumerator).toLowerCase() === enumeratorName.toLowerCase();
    const sameMobile = clean(official.enumerator_mobile) === mobile;

    if (!sameName || !sameMobile) {
      return json({
        success:false,
        error:"Enumerator details do not match the official Final List assignment for HLB " + hlb + "."
      },403);
    }
  }

  const date = currentDate();

  const already = await db.prepare(`
    SELECT id FROM hlb_submissions
    WHERE report_date=? AND hlb=?
  `).bind(date,hlb).first();

  if (already) {
    return json({
      success:false,
      error:"HLB " + hlb + " has already been submitted today. A fresh submission will automatically open tomorrow."
    },409);
  }

  const createdAt = new Date().toISOString();

  try {
    await db.prepare(`
      INSERT INTO hlb_submissions
      (report_date,enumerator_name,mobile,hlb,buildings,houses,synced,status,created_at)
      VALUES(?,?,?,?,?,?,?,?,?)
    `).bind(
      date,
      enumeratorName,
      mobile,
      hlb,
      buildings,
      houses,
      synced,
      "SUBMITTED",
      createdAt
    ).run();
  } catch (e) {
    if (String(e?.message || "").toLowerCase().includes("unique")) {
      return json({
        success:false,
        error:"This HLB has already been submitted today."
      },409);
    }
    throw e;
  }

  return json({
    success:true,
    message:"HLB " + hlb + " submitted successfully for " + displayDate(date) + ".",
    report_date:date,
    assignment:assignmentView(official)
  });
}

async function dailyReport(db,date) {
  const r = await db.prepare(`
    SELECT * FROM hlb_submissions
    WHERE report_date=?
    ORDER BY CAST(hlb AS INTEGER)
  `).bind(date).all();

  const rows = r.results || [];
  const byHLB = new Map();

  let buildings=0, houses=0, synced=0;

  for (const x of rows) {
    const h = normalizeHLB(x.hlb) || x.hlb;
    const b = num(x.buildings);
    const hs = num(x.houses);
    const sy = num(x.synced);

    buildings += b;
    houses += hs;
    synced += sy;

    byHLB.set(h,{
      id:x.id,
      report_date:x.report_date,
      hlb:h,
      enumerator_name:x.enumerator_name,
      mobile:x.mobile,
      buildings:b,
      houses:hs,
      synced:sy,
      percentage:pct(sy,hs),
      status:x.status,
      created_at:x.created_at
    });
  }

  return {
    success:true,
    report_date:date,
    report_date_display:displayDate(date),
    total_hlb:TOTAL_HLB,
    submitted_hlb:rows.length,
    pending_hlb:TOTAL_HLB-rows.length,
    buildings,
    houses,
    synced,
    percentage:pct(synced,houses),
    hlbs:allHLBs().map(h => byHLB.get(h) || {
      id:null,
      report_date:date,
      hlb:h,
      enumerator_name:"",
      mobile:"",
      buildings:0,
      houses:0,
      synced:0,
      percentage:0,
      status:"PENDING",
      created_at:""
    })
  };
}

async function overallReport(db) {
  // Latest submission per HLB is used for the current overall snapshot.
  // This avoids double-counting the same HLB across different dates.
  const r = await db.prepare(`
    SELECT s.*
    FROM hlb_submissions s
    INNER JOIN (
      SELECT hlb, MAX(id) AS max_id
      FROM hlb_submissions
      GROUP BY hlb
    ) latest
    ON s.id=latest.max_id
    ORDER BY CAST(s.hlb AS INTEGER)
  `).all();

  const latest = r.results || [];

  let buildings=0, houses=0, synced=0;

  for (const x of latest) {
    buildings += num(x.buildings);
    houses += num(x.houses);
    synced += num(x.synced);
  }

  const uniqueSubmitted = latest.length;

  const daysResult = await db.prepare(`
    SELECT
      report_date,
      COUNT(*) AS submitted_hlb,
      SUM(buildings) AS buildings,
      SUM(houses) AS houses,
      SUM(synced) AS synced
    FROM hlb_submissions
    GROUP BY report_date
    ORDER BY report_date DESC
  `).all();

  const days = (daysResult.results || []).map(x => ({
    date:x.report_date,
    submitted_hlb:num(x.submitted_hlb),
    buildings:num(x.buildings),
    houses:num(x.houses),
    synced:num(x.synced),
    percentage:pct(num(x.synced),num(x.houses))
  }));

  return {
    success:true,
    total_hlb:TOTAL_HLB,
    unique_hlbs_submitted:uniqueSubmitted,
    remaining_hlb:TOTAL_HLB-uniqueSubmitted,
    current_buildings:buildings,
    current_houses:houses,
    current_synced:synced,
    current_percentage:pct(synced,houses),
    days,
    last_report_date:days.length ? days[0].date : ""
  };
}

const encoder = new TextEncoder();

function b64url(bytes) {
  let s="";
  for (const b of bytes) s += String.fromCharCode(b);
  return btoa(s).replace(/\+/g,"-").replace(/\//g,"_").replace(/=+$/,"");
}

function fromB64url(s) {
  s=s.replace(/-/g,"+").replace(/_/g,"/");
  while (s.length % 4) s += "=";
  const bin=atob(s);
  const out=new Uint8Array(bin.length);
  for(let i=0;i<bin.length;i++) out[i]=bin.charCodeAt(i);
  return out;
}

async function signature(secret,data) {
  const key=await crypto.subtle.importKey(
    "raw",
    encoder.encode(secret),
    {name:"HMAC",hash:"SHA-256"},
    false,
    ["sign","verify"]
  );
  const sig=await crypto.subtle.sign("HMAC",key,encoder.encode(data));
  return b64url(new Uint8Array(sig));
}

async function verifySignature(secret,data,sig) {
  const key=await crypto.subtle.importKey(
    "raw",
    encoder.encode(secret),
    {name:"HMAC",hash:"SHA-256"},
    false,
    ["sign","verify"]
  );

  try {
    return await crypto.subtle.verify(
      "HMAC",
      key,
      fromB64url(sig),
      encoder.encode(data)
    );
  } catch {
    return false;
  }
}

function getCookie(request,name) {
  const cookie=request.headers.get("Cookie") || "";
  const parts=cookie.split(";");

  for(const p of parts) {
    const i=p.indexOf("=");
    if(i<0) continue;
    const k=p.slice(0,i).trim();
    if(k===name) return decodeURIComponent(p.slice(i+1));
  }

  return "";
}

async function makeSession(secret) {
  const payload={
    iat:Date.now(),
    exp:Date.now()+8*60*60*1000
  };

  const data=b64url(encoder.encode(JSON.stringify(payload)));
  return data + "." + await signature(secret,data);
}

async function isAdmin(request,env) {
  const secret=clean(env.ADMIN_SESSION_SECRET);
  if(!secret) return false;

  const token=getCookie(request,"admin_session");
  if(!token || !token.includes(".")) return false;

  const i=token.lastIndexOf(".");
  const data=token.slice(0,i);
  const sig=token.slice(i+1);

  if(!(await verifySignature(secret,data,sig))) return false;

  try {
    const payload=JSON.parse(new TextDecoder().decode(fromB64url(data)));
    return Number(payload.exp) > Date.now();
  } catch {
    return false;
  }
}

async function adminLogin(request,env) {
  const username=clean(env.ADMIN_USERNAME || env.ADMIN_USER);
  const password=clean(env.ADMIN_PASSWORD);
  const secret=clean(env.ADMIN_SESSION_SECRET);

  if(!username || !password || !secret) {
    return json({
      success:false,
      error:"Admin authentication is not configured. Add ADMIN_USERNAME, ADMIN_PASSWORD and ADMIN_SESSION_SECRET in Cloudflare Worker Secrets."
    },500);
  }

  let body;
  try {
    body=await request.json();
  } catch {
    return json({success:false,error:"Invalid login request."},400);
  }

  if(
    clean(body.username)!==username ||
    String(body.password ?? "")!==password
  ) {
    return json({success:false,error:"Invalid username or password."},401);
  }

  const token=await makeSession(secret);

  return json(
    {success:true,message:"Admin login successful."},
    200,
    {
      "set-cookie":
        "admin_session=" +
        encodeURIComponent(token) +
        "; Path=/; HttpOnly; Secure; SameSite=Lax; Max-Age=28800"
    }
  );
}

function adminLogout() {
  return json(
    {success:true,message:"Logged out."},
    200,
    {
      "set-cookie":
        "admin_session=; Path=/; HttpOnly; Secure; SameSite=Lax; Max-Age=0"
    }
  );
}

function esc(s){
 return String(s??"").replace(/[&<>"']/g,m=>({
  "&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#039;"
 }[m]));
}

async function doLogin(){
 const b=$("loginBtn");
 b.disabled=true;
 b.textContent="AUTHENTICATING...";

 try{
  const r=await fetch("/api/admin/login",{
   method:"POST",
   headers:{"content-type":"application/json"},
   credentials:"same-origin",
   body:JSON.stringify({
    username:$("u").value.trim(),
    password:$("p").value
   })
  });

  const d=await r.json();

  if(!r.ok||!d.success) throw new Error(d.error||"Login failed");

  await dashboard();
 }catch(e){
  $("loginMsg").innerHTML='<div class="error" style="margin-top:13px">'+esc(e.message)+'</div>';
 }finally{
  if($("loginBtn")){
   b.disabled=false;
   b.textContent="🔐 LOGIN TO ADMIN DASHBOARD";
  }
 }
}

async function api(path){
 const r=await fetch(path,{
  cache:"no-store",
  credentials:"same-origin"
 });

 let d={};
 try{d=await r.json()}catch{}

 if(r.status===401){
  logged=false;
  loginPage();
  throw new Error("Admin session expired. Please login again.");
 }

 if(!r.ok||!d.success) throw new Error(d.error||"Request failed");
 return d;
}

async function logout(){
 await fetch("/api/admin/logout",{
  method:"POST",
  credentials:"same-origin"
 });
 logged=false;
 loginPage();
}

async function doSearch(){
 try{
  const value=$("hlbSearch").value.trim();
  const d=await api("/api/admin/hlb?hlb="+encodeURIComponent(value));

  if(!d.assignment){
   $("searchResult").innerHTML='<div class="error">HLB '+esc(value)+' was not found in the official Final List.</div>';
   return;
  }

  const a=d.assignment;

  $("searchResult").innerHTML=\`
   <div class="detailgrid">
    <div class="detail"><span>HLB NUMBER</span><b>\${esc(a.hlb)}</b></div>
    <div class="detail"><span>SUPERVISORY CIRCLE NO</span><b>\${esc(a.supervisory_circle)}</b></div>

    <div class="detail" style="grid-column:1/-1"><span>VILLAGE NAME — OFFICIAL ALLOTMENT</span><b>\${esc(a.village_name||"")}</b></div>
   </div>

   <div style="margin-top:18px;font-weight:1000">📅 Submission History</div>

   <div class="tablewrap">
    <table class="table">
     <thead><tr><th>DATE</th><th>BUILDINGS</th><th>HOUSES</th><th>SYNCED</th><th>PROGRESS</th><th>STATUS</th></tr></thead>
     <tbody>
      \${d.history.length?d.history.map(x=>\`

        <td>\${esc(x.report_date)}</td>
